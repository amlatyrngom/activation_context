    30	def set_marks(root: Path, at3_id: str, tag: str, seeds: int) -> dict:
    31	    sample = json.loads((root / "PAPER_TRAJ_QA" / at3_id / "sample.json").read_text())["sample"]["deepmath"]
    32	    rows = load_solvers(root, at3_id)
    33	    tasks = [t for t in sample["task_ids"] if all((t, s) in rows for s in range(3))]
    34	    out = {}
    35	    for t in tasks:
    36	        task = rows[(t, 0)]["agent_config"]["dataset_task"]
    37	        answers = [solver_result(rows[(t, s)], 40_000) for s in range(3)]
    38	        solved = [score(task, a) >= 1.0 for a in answers]
    39	        majority = lenient.majority_answer("deepmath", answers)
    40	        strict = majority is not None and score(task, majority) >= 1.0
    41	        valid = [ok for answer, ok in zip(answers, solved) if isinstance(answer, str)]
    42	        tied = majority is None and bool(valid)
    43	        out[t] = {"diff": float(t.split("-")[0][1:]), "single": solved, "any@3": [any(solved)], "majority strict": [strict],
    44	                  "majority": [sum(valid) / len(valid) if tied else strict],
    45	                  "majority draw": [random.Random(t).choice(valid) if tied else strict]}
    46	    for name, kind, pattern in ARMS:
    47	        for s in range(seeds):
    48	            got = load(f"{kind}:{root / pattern.format(p=tag, s=s)}")
    49	            for t in tasks:
    50	                row = got.get(t)
    51	                ok = False
    52	                if row is not None and not row["bad"] and not row["length"]:
    53	                    answer, _ = lenient.readings("deepmath", row["text"])["lenient"]
    54	                    ok = score(row["task"], answer) >= 1.0
    55	                out[t].setdefault(name, []).append(ok)
    56	    print(f"{at3_id}: {len(tasks)} tasks", flush=True)
    57	    return out

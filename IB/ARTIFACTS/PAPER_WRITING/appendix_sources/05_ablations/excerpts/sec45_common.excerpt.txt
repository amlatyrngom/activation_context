    69	def linear_skip() -> bool:
    70	    """v7 (Oct 27): SEC45_LINEAR_SKIP=1 starts a FRESH smaller-encoder pair with the row head's linear skip (row_head_linear_skip: rows
    71	    start as the least-squares side->reader embedding map of the encoder's final states instead of a random FFN projection). Only with
    72	    SEC45_ENCODER (equal widths use the identity skip). Unset = v6 behaviour. A loaded or resumed pair keeps the head it was saved with."""
    73	    on = os.environ.get("SEC45_LINEAR_SKIP") == "1"
    74	    if on and not os.environ.get("SEC45_ENCODER"):
    75	        raise SystemExit("SEC45_LINEAR_SKIP=1 needs SEC45_ENCODER (a smaller encoder; equal widths use the identity skip)")
    76	    return on

   123	def recon_items(study, passages: list[dict], seed: int = 0):
   124	    """Plain passage reconstruction at 1/10: the passage is one AC part, the target is the whole passage (SFT)."""
   125	    rows = [{"text": p["text"], "doc_id": p["doc_id"], "source": p.get("source", "fineweb_edu")} for p in passages]

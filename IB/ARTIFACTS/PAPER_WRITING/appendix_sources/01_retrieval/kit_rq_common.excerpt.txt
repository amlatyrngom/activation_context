    91	def inject_golds(top: dict, questions: list[dict], n: int) -> dict:
    92	    """id -> the first `n` of the BM25 ranking with every gold present (user, Oct 21: "Yes on injecting the golds"): each gold
    93	    missing from the first n replaces a non-gold passage at a position drawn from crc32(id/gold), so a gold sits at a seeded
    94	    rank among the retrieved ones (the order seen by the reader carries no gold signal); retrieved golds keep their rank."""
    95	    import zlib
    96	    out = {}
    97	    for q in questions:
    98	        ranking = [int(x) for x in top[q["id"]][:n]]
    99	        golds = [int(g) for g in q["golds"]]
   100	        for gold in golds:
   101	            if gold in ranking:
   102	                continue
   103	            free = [i for i, x in enumerate(ranking) if x not in golds]
   104	            ranking[free[zlib.crc32(f"{q['id']}/{gold}".encode()) % len(free)]] = gold
   105	        assert set(golds) <= set(ranking) and len(ranking) == len(set(ranking)) == min(n, len(top[q["id"]]))
   106	        out[q["id"]] = ranking
   107	    return out

   190	def build_items(study, corpus, questions: list[dict], top: dict, ac_name: str, ratio: float, *, replies: dict | None = None,
   191	                calls: int = 20, per_call: int = 20, kind: str = "rag_qa", question_first: bool = False, same_context: bool = False):
   192	    """ActivationContextTrainingItems. With `replies` (id -> thinking reply text), the target is that reply and the teacher view
   193	    is the gold-passage prompt (KL); without, the target is a placeholder (generation only: the decode reads the prefix)."""
   194	    from activation.ac_model.ac_model_training import ActivationContextTrainingItem
   195	    from activation.ac_model.ac_model_utils import assistant_target_ids
   196	    items = []
   197	    for q in questions:
   198	        reply = split_reply(replies[q["id"]]) if replies is not None else ("", "Final answer: ?")
   199	        if reply is None:
   200	            continue
   201	        output = {"role": "assistant", "content": reply[1], **({"reasoning_content": reply[0]} if reply[0] else {})}
   202	        ids = assistant_target_ids(study.tokenizer, output, None, THINKING)

   206	            in_context_start=1, ac_start=1, assistant_token_ids=[ids], dataset_id="hotpotqa",
   207	            info={"question": q["question"], "answer": q["answer"], "ratio": ratio, "calls": calls, "per_call": per_call, "question_first": question_first,
   208	                  "teacher_view": "same_context" if same_context else "gold",
   209	                  "chat_template_kwargs": dict(THINKING)}))

    80	    The training recipe. The defaults are the ones the reconstruction bench settled on: SFT on the gold target tokens, cosine
    81	    decay with a linear warm-up, one joint gradient-norm clip over the encoder and the reader adapter, the no-context penalty
    82	    (λ x KL(base || adapted reader) on the same history with its AC parts removed, at the target positions, on a random
    83	    quarter of the examples per update and scaled by 1 / fraction) whenever the reader adapter trains, and stored teacher /
    84	    reference targets that always hold the gold token. `learning_rate_target_lora = 0` freezes the reader adapter (it neither
    85	    steps nor carries the penalty): the regime that adapts the encoder alone to a dataset or task.
    86	    """
    87	    loss_kind: t.Literal["kl", "sft"] = "sft"   # "sft": cross-entropy on the target tokens; "kl": the coarsened KL to the stored teacher (the reader as it is at the
    88	                                               # call's start, on the in-context history; its top-k log-probs + the remainder mass are stored before the first update)
    89	    learning_rate_ac: float = 1e-4             # AC modules and the side LoRA
    90	    learning_rate_target_lora: float = 2e-5    # the target adapter (0 = frozen: no steps, no penalty)

    96	    lr_schedule: str = "cosine"                # "constant" (warm-up only), "cosine": decay to lr_final_fraction of the base rate over the planned updates, or "cooldown" (warm-up → constant → decay over the last cooldown_updates), or "cyclical" (that, repeated every cycle_updates)

   102	    cooldown_updates: int | None = None        # lr_schedule "cooldown": the decay (to lr_final_fraction) spans the last cooldown_updates of the planned updates; constant before

   106	    cycle_reset_optimizer: bool = False        # lr_schedule "cyclical": the Adam moments and step counts start fresh at every cycle start after the first, so a cycle is a restart from the pair the cycle before ended on

   109	    checkpoint_every_updates: int | None = None  # mid-epoch checkpoint (AC_MODELS/<name>/epoch_<n>_step_<s> + optimizer.pt + partial_epoch.json) every this many steps
